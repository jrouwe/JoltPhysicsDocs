229 {
230 float dist_sq = inWorldSpaceBounds.GetSqDistanceTo(inCameraPosition);
231 for (const LOD &lod : mLODs)
232 if (dist_sq <= inLODScaleSq * Square(lod.mDistance))
233 return lod;
234
235 return mLODs.back();
236 }
